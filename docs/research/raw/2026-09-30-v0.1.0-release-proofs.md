## mcpdrive.py

Condensed from the script that ran: the same messages and prints; the run's version also
exits with the server's stderr when stdout closes early.

```python
import json, subprocess, sys, threading

p = subprocess.Popen(sys.argv[1:], stdin=subprocess.PIPE, stdout=subprocess.PIPE, stderr=subprocess.PIPE, text=True)
err_lines = []
threading.Thread(target=lambda: err_lines.extend(p.stderr), daemon=True).start()

def send(msg):
    p.stdin.write(json.dumps(msg) + "\n"); p.stdin.flush()

def recv(want_id):
    while True:
        msg = json.loads(p.stdout.readline())
        if msg.get("id") == want_id:
            return msg

send({"jsonrpc": "2.0", "id": 1, "method": "initialize", "params": {
    "protocolVersion": "2025-11-25", "capabilities": {}, "clientInfo": {"name": "release-proof", "version": "0"}}})
init = recv(1)
print("initialize.result.protocolVersion:", init["result"]["protocolVersion"])
print("initialize.result.serverInfo:", json.dumps(init["result"]["serverInfo"]))
send({"jsonrpc": "2.0", "method": "notifications/initialized"})
send({"jsonrpc": "2.0", "id": 2, "method": "tools/list"})
print("tools/list names:", [t["name"] for t in recv(2)["result"]["tools"]])
p.stdin.close(); p.wait(timeout=10)
print("exit code:", p.returncode)
for line in err_lines:
    if '"server started"' in line:
        print("stderr server started:", line.strip())
```
